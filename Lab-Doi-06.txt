drawVerticalLine(targetTime, colorTime) =>
    line.new(x1 = targetTime, y1 = low, x2 = targetTime, y2 = high, xloc = xloc.bar_time, extend = extend.both, color = colorTime, style = line.style_dashed, width = 1)
// HMA
hma(src, length) =>
    halfLength = math.round(length / 2)
    sqrtLength = math.round(math.sqrt(length))
    wma1 = ta.wma(src, halfLength)
    wma2 = ta.wma(src, length)
    hmaValue = ta.wma(2 * wma1 - wma2, sqrtLength)
    hmaValue

//###############################################################

var isEnd = bool(false)
//--------------------------------

//###############################################################

//###############################################################

HMA_ColorDw = #FF0000
NotDownTrendColor=#00FFFF
AllTimeHighColor=#FFAA00
BanBG=#666666
CDCcolorUp = #00FF0066
CDCcolorDw = #FF000066
BINANCE_FEE_RATE = float(0.04) // 0.02 for buy + 0.02 for sell 
//--------------------------------
var isLongOpen = bool(false)
var Long_AvgPrice = float(0) // USDT
var Long_SumOfBTC = float(0) // Coin
var Long_MaxDD = float(0)

var Long_AccumulateProfit = float(0)
var Long_WinCount = int(0)

var doBuy_Short = bool(false)

doBuy_Short := false
doSell_Short := false

var doSell_Long2 = bool(false)

doSell_Long2 := false

doDCA_Long := false
var doSellDCA_Long = bool(false)
doSellDCA_Long := false

var isBuyLongBack=bool(false)

isOverSold:=false

        NotDownTrend:=HighestLine
    if NotDownTrend<LowestLine
        NotDownTrend:=LowestLine

    if NotDownTrend<LowestLine
        NotDownTrend:=LowestLine

        NotDownTrend:=LowestLine_AllTimeHigh

HMAs_Line0 = request.security(syminfo.tickerid,"30",hma(close, 21))
HMAs_Line = ta.ema(HMAs_Line0,30)

if HMAs_Line>LowestLine_AllTimeHigh
    HMAs_LineColor :=  HMA_ColorNormal
else
    if HMAs_Line > HMAs_Line[1]
        HMAs_LineColor :=  HMA_ColorUp

else

        HMA_LineColor :=  HMA_ColorUp
    else
        if HMA_Line < HMA_Line[1] and HMAs_Line < HMAs_Line[1] 
            HMA_LineColor := HMA_ColorDw
        else
            HMA_LineColor :=  HMA_ColorNormal

CDC_maFast = request.security(syminfo.tickerid, "720",ta.ema(close, 12))

CDC_Color := CDC_maFast > CDC_maSlow ? CDCcolorUp : CDCcolorDw

RSI_up = ta.rma(math.max(RSI_change, 0), 14)
RSI_down = ta.rma(-math.min(RSI_change, 0), 14)
RSI = RSI_down == 0 ? 100 : RSI_up == 0 ? 0 : 100 - 100 / (1 + RSI_up / RSI_down)
if RSI>70 or RSI[1]>70
    if RSI[1]>RSI
        isOverBought:=true
if RSI<30 or RSI[1]<30
    if RSI[1]<RSI
        isOverSold:=true

                Long_MaxSumOfBTC:=Long_SumOfBTC
            isLongOpen := true
            if BUY_SIDE == 'Long'
                strategy.entry('L', strategy.long, qty = Long_SumOfBTC, comment = str.tostring(Long_SumOfBTC))
            doBuy_Long := true
            isBuyLongBack:=true

// Strategy : DCA More Long When Loss

//---------------------------------------------------------------
if isActive and isEnd==false and barstate.isconfirmed 
    if close<Long_AvgPrice and Long_isDaysPassed and isBuyPoint and isLongOpen==true 
        //------------------------------
        // Open Long
        //------------------------------
        Long_AvgPrice := ((close*LONG_POSITION_SIZE)+(Long_AvgPrice*Long_SumOfBTC))/(Long_SumOfBTC+LONG_POSITION_SIZE)
        Long_SumOfBTC := Long_SumOfBTC+LONG_POSITION_SIZE
        isLongOpen := true
        if BUY_SIDE == 'Long'
            strategy.entry('L', strategy.long, qty = LONG_POSITION_SIZE, comment = str.tostring(LONG_POSITION_SIZE))
        doDCA_Long := true
        Long_LastBuyTime := time

if isActive and isEnd==false and barstate.isconfirmed 
    if close>Long_AvgPrice and isSellPoint and isLongOpen==true
        diff = Long_SumOfBTC-LONG_MIN_POSITION_SIZE
        if diff>0
            //------------------------------
            // Close Long
            //------------------------------
            if isLongOpen==true and Long_SumOfBTC>0 
                // Save Profit -------------
                TmpFee=((BINANCE_FEE_RATE*close*diff)/100)+((BINANCE_FEE_RATE*Long_AvgPrice*diff)/100)
                Long_TmpProfit := (close*diff)-(Long_AvgPrice*diff)-TmpFee
                Long_AccumulateProfit := Long_AccumulateProfit + Long_TmpProfit
                if Long_TmpProfit>0
                    Long_WinCount := Long_WinCount + 1
                    Long_GrossWin := Long_GrossWin + Long_TmpProfit
                else
                    Long_LossCount := Long_LossCount + 1
                    Long_GrossLoss := Long_GrossLoss + Long_TmpProfit
                // Close Long -------------
                if BUY_SIDE == 'Long'
                    strategy.close('L', qty = diff, comment = str.tostring(diff))
                Long_SumOfBTC := Long_SumOfBTC-diff
                CurrentCapital:=CurrentCapital+Long_TmpProfit
                doSellDCA_Long := true

            Short_SumOfBTC := 0
