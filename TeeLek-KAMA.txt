//@version=6
indicator('TeeLek KAMA', overlay = true)

//###############################################################
// Input
//###############################################################
IS_SHOW_DAY = input.bool(true,"Show day")
IS_SHOW_12H = input.bool(true,"Show 12 hr")
IS_SHOW_6H = input.bool(true,"Show 6 hr")
IS_SHOW_4H = input.bool(true,"Show 4 hr")
IS_SHOW_2H = input.bool(true,"Show 2 hr")
IS_SHOW_1H = input.bool(true,"Show 1 hr")

//###############################################################
// KAMA
//###############################################################
KAMA_PriceChange = math.abs(close - close[10])
KAMA_PriceChangeSum = math.sum(math.abs(close - close[1]), 10)
KAMA_Fastest = 2/6
KAMA_Slowest = 2/51
KAMA_ER = KAMA_PriceChange / KAMA_PriceChangeSum
KAMA_SC = math.pow(KAMA_ER * (KAMA_Fastest - KAMA_Slowest) + KAMA_Slowest, 2)
KAMA_KAMA = 0.0
KAMA_KAMA := na(KAMA_KAMA[1]) ? close : KAMA_KAMA[1] + KAMA_SC * (close - nz(KAMA_KAMA[1]))
//-----------------------------------------------------------------------------
KAMA_CloseLinePlot = plot(close, '', display = display.none)
//-----------------------------------------------------------------------------

colorUp = #9cff87
colorDw = #f9396a

//###############################################################
// KAMA : DAY
//###############################################################
KAMADAY_KAMA0 = request.security(syminfo.tickerid, "D", KAMA_KAMA)
KAMADAY_KAMA = ta.ema(KAMADAY_KAMA0,3)
KAMADAY_Plot = plot(KAMADAY_KAMA, title = 'KAMA', color = IS_SHOW_DAY ? close>KAMADAY_KAMA ? colorUp : colorDw : na, linewidth=1)
fill(KAMADAY_Plot, KAMA_CloseLinePlot, math.max(KAMADAY_KAMA, close), KAMADAY_KAMA, IS_SHOW_DAY ? color.new(colorUp, 75) : na, IS_SHOW_DAY ? color.new(colorUp, 60) : na)
fill(KAMADAY_Plot, KAMA_CloseLinePlot, KAMADAY_KAMA, math.min(KAMADAY_KAMA, close), IS_SHOW_DAY ? color.new(colorDw, 60) : na, IS_SHOW_DAY ? color.new(colorDw, 75) : na)
