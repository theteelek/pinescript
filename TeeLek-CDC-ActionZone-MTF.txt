// Week ###########################################################
CDCWeek_maFast0 = request.security(syminfo.tickerid, "W",ta.ema(close, 12))
CDCWeek_maSlow0 = request.security(syminfo.tickerid, "W",ta.ema(close, 26))
CDCWeek_maFast = ta.sma(CDCWeek_maFast0,3)
CDCWeek_maSlow = ta.sma(CDCWeek_maSlow0,3)
var CDCWeek_Color = color(na)
if CDCWeek_maFast > CDCWeek_maSlow
    CDCWeek_Color := #00FF0066
else
    CDCWeek_Color := #FF000066
CDCWeek_FastPlot = plot(CDCWeek_maFast, color = na, linewidth = 1)
CDCWeek_SlowPlot = plot(CDCWeek_maSlow, color = na, linewidth = 1)
fill(CDCWeek_FastPlot,CDCWeek_SlowPlot,color=IS_SHOW_WEEK?CDCWeek_Color:na)
