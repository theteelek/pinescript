//@version=6
strategy('Lab-Doi-05', overlay = true, pyramiding=2000)
BUY_SIDE = 'None'

//###############################################################
// Input
//###############################################################
DATE_START = input.time(timestamp('16 Dec 2024 01:00'), title = 'Start Date')
DATE_END = input.time(timestamp('5 Jun 2025 14:00'), title = 'Start Date')
//--------------------------------
var LONG_MIN_POSITION_SIZE = float(0.10) // Coin
var LONG_MIN_POSITION_SIZE_HALF = float(0.05) // Coin
//--------------------------------
INTITIAL_CAPITAL = 10000 // USD
//--------------------------------

//###############################################################
// Function
//###############################################################
drawVerticalLine(targetTime, colorTime) =>
    line.new(x1 = targetTime, y1 = low, x2 = targetTime, y2 = high, xloc = xloc.bar_time, extend = extend.both, color = colorTime, style = line.style_dashed, width = 1)

//###############################################################
// Variable
//###############################################################
var isActive = bool(false)
var isEnd = bool(false)
//--------------------------------
var isLongOpen = bool(false)
var Long_AvgPrice = float(0) // USDT
var Long_SumOfBTC = float(0) // Coin
var Long_MaxDD = float(0)
var Long_TmpProfit=float(0)
var Long_AccumulateProfit = float(0)
var Long_WinCount = int(0)
var Long_LossCount = int(0)
var Long_GrossWin = float(0)
var Long_GrossLoss = float(0)
//--------------------------------
var isShortOpen = bool(false)
var Short_AvgPrice = float(0) // USDT
var Short_SumOfBTC = float(0) // Coin
var Short_MaxDD = float(0)
var Short_TmpProfit = float(0)
var Short_AccumulateProfit = float(0)
var Short_WinCount = int(0)
var Short_LossCount = int(0)
var Short_GrossWin = float(0)
var Short_GrossLoss = float(0)
//--------------------------------
var isUseLong2 = bool(false)
var isLong2Open = bool(false)
var Long2_AvgPrice = float(0) // USDT
var Long2_SumOfBTC = float(0) // Coin
var Long2_MaxDD = float(0)
var Long2_TmpProfit=float(0)
var Long2_AccumulateProfit = float(0)
var Long2_WinCount = int(0)
var Long2_LossCount = int(0)
var Long2_GrossWin = float(0)
var Long2_GrossLoss = float(0)
//--------------------------------
var CurrentCapital = float(0)
if CurrentCapital==0
    CurrentCapital:=INTITIAL_CAPITAL

//###############################################################
// Config
//###############################################################
CDCcolorUp = #00FF0066
CDCcolorDw = #FF000066
BINANCE_FEE_RATE = float(0.04) // 0.02 for buy + 0.02 for sell 

//###############################################################
// Signal
//###############################################################
CDC_maFast = request.security(syminfo.tickerid, "D",ta.ema(close, 12))
CDC_maSlow = request.security(syminfo.tickerid, "D",ta.ema(close, 26))
var CDC_Color = color(na)
CDC_Color := CDC_maFast > CDC_maSlow ? CDCcolorUp : CDCcolorDw

//###############################################################
// Calculate
//###############################################################
if time >= DATE_START 
    isActive := true
else
    isActive := false
if time >= DATE_END
    isEnd := true
else
    isEnd := false

//###############################################################
// Strategy : Open First Long
//###############################################################
if isActive and isLongOpen==false and CDC_maFast>CDC_maSlow and barstate.isconfirmed 
    //------------------------------
    // Open Long
    //------------------------------
    if isLongOpen==false and Long_SumOfBTC==0
        Long_AvgPrice := close
        Long_SumOfBTC := LONG_MIN_POSITION_SIZE
        isLongOpen := true
        if BUY_SIDE == 'Long'
            strategy.entry('L', strategy.long, qty = Long_SumOfBTC, comment = str.tostring(Long_SumOfBTC))

//###############################################################
// Strategy : Open Short
//###############################################################
if isActive and CDC_maFast<CDC_maSlow and isShortOpen==false and barstate.isconfirmed 
    //------------------------------
    // Open Short
    //------------------------------
    if isShortOpen==false and Short_SumOfBTC==0
        Short_AvgPrice := close
        // Short_SumOfBTC := LONG_MIN_POSITION_SIZE_HALF
        Short_SumOfBTC := LONG_MIN_POSITION_SIZE
        isShortOpen := true
        if BUY_SIDE == 'Short'
            strategy.entry('S', strategy.short, qty = Short_SumOfBTC, comment = str.tostring(Short_SumOfBTC))
        isUseLong2:=true

//###############################################################
// Strategy : Close Short
//###############################################################
if isActive and CDC_maFast>CDC_maSlow and isShortOpen==true and barstate.isconfirmed 
    //------------------------------
    // Close Short
    //------------------------------
    if isShortOpen==true and Short_SumOfBTC>0
        // Save Profit -------------
        TmpFee=((BINANCE_FEE_RATE*close*Short_SumOfBTC)/100)+((BINANCE_FEE_RATE*Short_AvgPrice*Short_SumOfBTC)/100)
        Short_TmpProfit := (Short_AvgPrice*Short_SumOfBTC)-(close*Short_SumOfBTC)-TmpFee
        Short_AccumulateProfit := Short_AccumulateProfit + Short_TmpProfit
        if Short_TmpProfit>0
            Short_WinCount := Short_WinCount + 1
            Short_GrossWin := Short_GrossWin + Short_TmpProfit
        else
            Short_LossCount := Short_LossCount + 1
            Short_GrossLoss := Short_GrossLoss + Short_TmpProfit
        // Close Short -------------
        if BUY_SIDE == 'Short'
            strategy.close('S', qty = Short_SumOfBTC, comment = str.tostring(Short_SumOfBTC))

        Short_SumOfBTC := 0
        isShortOpen := false

//###############################################################

    //------------------------------
    // Close Long2
    //------------------------------
    if isLong2Open==true and Long2_SumOfBTC>0 and close<Long2_AvgPrice 
        // Save Profit -------------
        TmpFee=((BINANCE_FEE_RATE*close*LONG_MIN_POSITION_SIZE)/100)+((BINANCE_FEE_RATE*Long2_AvgPrice*LONG_MIN_POSITION_SIZE)/100)
        Long2_TmpProfit := (close*LONG_MIN_POSITION_SIZE)-(Long2_AvgPrice*LONG_MIN_POSITION_SIZE)-TmpFee
        Long2_AccumulateProfit := Long2_AccumulateProfit + Long2_TmpProfit
        if Long2_TmpProfit>0
            Long2_WinCount := Long2_WinCount + 1
            Long2_GrossWin := Long2_GrossWin + Long2_TmpProfit
        else
            Long2_LossCount := Long2_LossCount + 1
            Long2_GrossLoss := Long2_GrossLoss + Long2_TmpProfit
        // Close Long2 -------------
        if BUY_SIDE == 'Long2'
            strategy.close('L', qty = LONG_MIN_POSITION_SIZE, comment = str.tostring(LONG_MIN_POSITION_SIZE))
        Long2_SumOfBTC := 0
        isLong2Open:=false
        Long2_AvgPrice:=0
        CurrentCapital:=CurrentCapital+Long2_TmpProfit
        isUseLong2:=true

//###############################################################
// Drawing

            Short_MaxDD:=Short_TmpProfit

        Long2_TmpProfit := (close*Long2_SumOfBTC)-(Long2_AvgPrice*Long2_SumOfBTC)

        if Long2_MaxDD==0

        if Long2_MaxDD>Long2_TmpProfit
            Long2_MaxDD:=Long2_TmpProfit
//--------------------------------------
TotalProfit=Long_GrossWin+Long_GrossLoss+Long_TmpProfit+Short_GrossWin+Short_GrossLoss+Short_TmpProfit+Long2_GrossWin+Long2_GrossLoss+Long2_TmpProfit
